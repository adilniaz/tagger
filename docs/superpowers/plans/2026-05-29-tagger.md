# Tagger Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Python TUI app using Textual for browsing, viewing, and editing metadata tags on audio and ebook files.

**Architecture:** A Textual `App` drives multiple `Screen` classes. All file I/O is isolated in `core/tag_io.py` which exposes plain `read(path) -> dict` / `write(path, dict)` functions; the UI never imports mutagen or ebook libraries directly. Settings (which extended fields to show) live in `core/settings.py` and persist to `~/.config/tagger/settings.json`.

**Tech Stack:** Python 3.11+, textual>=0.60, mutagen, ebooklib, pypdf, pytest, pytest-asyncio

```python
from __future__ import annotations
import json
from dataclasses import dataclass, field
from pathlib import Path

CONFIG_PATH = Path.home() / ".config" / "tagger" / "settings.json"

EXTENDED_FIELDS = ["albumartist", "discnumber", "composer", "comment", "lyrics"]


@dataclass
class Settings:
    enabled_extended: set[str] = field(default_factory=set)

    @classmethod
    def load(cls) -> Settings:
        if not CONFIG_PATH.exists():
            return cls()
        data = json.loads(CONFIG_PATH.read_text())
        return cls(enabled_extended=set(data.get("extended_fields", [])))

    def save(self) -> None:
        CONFIG_PATH.parent.mkdir(parents=True, exist_ok=True)
        CONFIG_PATH.write_text(
            json.dumps({"extended_fields": sorted(self.enabled_extended)}, indent=2)
        )
```

```python
from __future__ import annotations
from pathlib import Path
from mutagen import File as MutagenFile  # noqa: F401

AUDIO_EXTENSIONS = frozenset({
    ".mp3", ".m4a", ".mp4", ".flac", ".ogg", ".opus",
    ".wma", ".wav", ".aiff", ".aif", ".ape", ".wv", ".mpc",
})
EBOOK_EXTENSIONS = frozenset({".epub", ".pdf", ".fb2", ".mobi", ".azw"})
SUPPORTED_EXTENSIONS = AUDIO_EXTENSIONS | EBOOK_EXTENSIONS

BASIC_FIELDS = ["title", "artist", "album", "date", "tracknumber", "genre"]
EXTENDED_FIELDS = ["albumartist", "discnumber", "composer", "comment", "lyrics"]
ALL_FIELDS = BASIC_FIELDS + EXTENDED_FIELDS

FIELD_LABELS: dict[str, str] = {
    "title": "Title",
    "artist": "Artist",
    "album": "Album",
    "date": "Year",
    "tracknumber": "Track",
    "genre": "Genre",
    "albumartist": "Album Artist",
    "discnumber": "Disc",
    "composer": "Composer",
    "comment": "Comment",
    "lyrics": "Lyrics",
}


def read(path: Path) -> dict[str, str]:
    ext = path.suffix.lower()
    if ext in AUDIO_EXTENSIONS:
        return _read_audio(path)
    if ext in EBOOK_EXTENSIONS:
        return _read_ebook(path)
    raise ValueError(f"Unsupported format: {ext}")


def write(path: Path, tags: dict[str, str]) -> None:
    ext = path.suffix.lower()
    if ext in AUDIO_EXTENSIONS:
        _write_audio(path, tags)
    elif ext in EBOOK_EXTENSIONS:
        _write_ebook(path, tags)
    else:
        raise ValueError(f"Unsupported format: {ext}")


def _read_audio(path: Path) -> dict[str, str]:
    f = MutagenFile(str(path), easy=True)
    if f is None:
        return {}
    result: dict[str, str] = {}
    for key in ALL_FIELDS:
        val = f.get(key)
        if val:
            result[key] = str(val[0])
    return result


def _write_audio(path: Path, tags: dict[str, str]) -> None:
    f = MutagenFile(str(path), easy=True)
    if f is None:
        raise IOError(f"Cannot open audio file: {path}")
    for key, value in tags.items():
        if value:
            f[key] = value
        elif key in f:
            del f[key]
    f.save()


# Ebook backends — implemented in Task 4
def _read_ebook(path: Path) -> dict[str, str]:
    raise NotImplementedError

def _write_ebook(path: Path, tags: dict[str, str]) -> None:
    raise NotImplementedError
```

```python
def _read_ebook(path: Path) -> dict[str, str]:
    ext = path.suffix.lower()
    if ext == ".epub":
        return _read_epub(path)
    if ext == ".pdf":
        return _read_pdf(path)
    if ext == ".fb2":
        return _read_fb2(path)
    if ext in (".mobi", ".azw"):
        return _read_mobi(path)
    return {}


def _write_ebook(path: Path, tags: dict[str, str]) -> None:
    ext = path.suffix.lower()
    if ext == ".epub":
        _write_epub(path, tags)
    elif ext == ".pdf":
        _write_pdf(path, tags)
    elif ext == ".fb2":
        _write_fb2(path, tags)
    # mobi/azw: write not supported


def _read_epub(path: Path) -> dict[str, str]:
    from ebooklib import epub
    book = epub.read_epub(str(path))
    mapping = [
        ("title", "title"), ("artist", "creator"),
        ("album", "publisher"), ("date", "date"), ("genre", "subject"),
    ]
    result: dict[str, str] = {}
    for field_key, dc_key in mapping:
        items = book.get_metadata("DC", dc_key)
        if items:
            result[field_key] = str(items[0][0])
    return result


def _write_epub(path: Path, tags: dict[str, str]) -> None:
    from ebooklib import epub
    book = epub.read_epub(str(path))
    dc_map = {
        "title": "title", "artist": "creator",
        "album": "publisher", "date": "date", "genre": "subject",
    }
    for field_key, dc_key in dc_map.items():
        if field_key in tags:
            book.set_unique_metadata("DC", dc_key, tags[field_key])
    epub.write_epub(str(path), book)


def _read_pdf(path: Path) -> dict[str, str]:
    from pypdf import PdfReader
    reader = PdfReader(str(path))
    meta = reader.metadata
    mapping = [
        ("title", meta.title),
        ("artist", meta.author),
        ("genre", meta.subject),
        ("albumartist", meta.creator),
    ]
    return {k: v for k, v in mapping if v}


def _write_pdf(path: Path, tags: dict[str, str]) -> None:
    from pypdf import PdfReader, PdfWriter
    reader = PdfReader(str(path))
    writer = PdfWriter()
    for page in reader.pages:
        writer.add_page(page)
    writer.add_metadata({
        "/Title": tags.get("title", ""),
        "/Author": tags.get("artist", ""),
        "/Subject": tags.get("genre", ""),
        "/Creator": tags.get("albumartist", ""),
    })
    with open(str(path), "wb") as f:
        writer.write(f)


def _read_fb2(path: Path) -> dict[str, str]:
    import xml.etree.ElementTree as ET
    ns = {"fb": "http://www.gribuser.ru/xml/fictionbook/2.0"}
    try:
        tree = ET.parse(str(path))
        root = tree.getroot()
        title_info = root.find(".//fb:title-info", ns)
        if title_info is None:
            return {}
        result: dict[str, str] = {}
        bt = title_info.find("fb:book-title", ns)
        if bt is not None and bt.text:
            result["title"] = bt.text
        author = title_info.find("fb:author", ns)
        if author is not None:
            fn = author.find("fb:first-name", ns)
            ln = author.find("fb:last-name", ns)
            name = " ".join(
                p for p in [
                    fn.text if fn is not None else "",
                    ln.text if ln is not None else "",
                ] if p
            ).strip()
            if name:
                result["artist"] = name
        return result
    except ET.ParseError:
        return {}


def _write_fb2(path: Path, tags: dict[str, str]) -> None:
    import xml.etree.ElementTree as ET
    ns = {"fb": "http://www.gribuser.ru/xml/fictionbook/2.0"}
    try:
        tree = ET.parse(str(path))
        root = tree.getroot()
        title_info = root.find(".//fb:title-info", ns)
        if title_info is None:
            return
        if "title" in tags:
            bt = title_info.find("fb:book-title", ns)
            if bt is not None:
                bt.text = tags["title"]
        tree.write(str(path), xml_declaration=True, encoding="utf-8")
    except ET.ParseError:
        pass


def _read_mobi(path: Path) -> dict[str, str]:
    try:
        import mobi
        import glob
        import xml.etree.ElementTree as ET
        _, extracted_dir = mobi.extract(str(path))
        opf_files = glob.glob(f"{extracted_dir}/*.opf")
        if not opf_files:
            return {}
        tree = ET.parse(opf_files[0])
        root = tree.getroot()
        ns = {"dc": "http://purl.org/dc/elements/1.1/"}
        result: dict[str, str] = {}
        t = root.find(".//dc:title", ns)
        if t is not None and t.text:
            result["title"] = t.text
        a = root.find(".//dc:creator", ns)
        if a is not None and a.text:
            result["artist"] = a.text
        return result
    except Exception:
        return {}
```

```python
from textual import on
from core.tag_io import SUPPORTED_EXTENSIONS


class FileList(ListView):
    class FileSelected(Message):
        def __init__(self, path: Path) -> None:
            super().__init__()
            self.path = path

    def __init__(self, **kwargs: object) -> None:
        super().__init__(**kwargs)
        self._paths: list[Path] = []
        self._dirty: set[Path] = set()

    def load(self, path: Path) -> None:
        self._dirty.clear()
        if path.is_dir():
            self._paths = sorted(
                p for p in path.iterdir()
                if p.is_file() and p.suffix.lower() in SUPPORTED_EXTENSIONS
            )
        else:
            self._paths = [path]
        self._refresh()

    def mark_dirty(self, path: Path) -> None:
        self._dirty.add(path)
        self._refresh()

    def mark_clean(self, path: Path) -> None:
        self._dirty.discard(path)
        self._refresh()

    def has_dirty(self) -> bool:
        return bool(self._dirty)

    def all_paths(self) -> list[Path]:
        return list(self._paths)

    def _refresh(self) -> None:
        self.clear()
        if not self._paths:
            self.append(ListItem(Label("No supported files found")))
            return
        for path in self._paths:
            label = path.name + (" *" if path in self._dirty else "")
            self.append(ListItem(Label(label), name=str(path)))

    @on(ListView.Selected)
    def _on_selected(self, event: ListView.Selected) -> None:
        if event.item.name:
            self.post_message(self.FileSelected(Path(event.item.name)))
```

```python
from __future__ import annotations
from pathlib import Path
from textual.app import ComposeResult
from textual.message import Message
from textual.widget import Widget
from textual.widgets import Input, Label, Button
from textual.containers import VerticalScroll
from core.tag_io import read, write, BASIC_FIELDS, EXTENDED_FIELDS, FIELD_LABELS
from core.settings import Settings


class TagEditor(Widget):
    class FileSaved(Message):
        def __init__(self, path: Path) -> None:
            super().__init__()
            self.path = path

    class DirtyChanged(Message):
        def __init__(self, path: Path, dirty: bool) -> None:
            super().__init__()
            self.path = path
            self.dirty = dirty

    def __init__(self, **kwargs: object) -> None:
        super().__init__(**kwargs)
        self._path: Path | None = None
        self._original: dict[str, str] = {}

    def compose(self) -> ComposeResult:
        yield Label("No file loaded — press ^B to browse", id="te-placeholder")

    def load(self, path: Path) -> None:
        self._path = path
        try:
            self._original = read(path)
        except Exception as exc:
            self._original = {}
            self.query("*").remove()
            self.mount(Label(f"Error reading file: {exc}"))
            return
        self._rebuild(Settings.load())

    def reload_settings(self) -> None:
        if self._path:
            self._rebuild(Settings.load())

    def _rebuild(self, settings: Settings) -> None:
        self.query("*").remove()
        fields = BASIC_FIELDS + [
            f for f in EXTENDED_FIELDS if f in settings.enabled_extended
        ]
        scroll = VerticalScroll()
        self.mount(scroll)
        for field_key in fields:
            scroll.mount(Label(FIELD_LABELS[field_key]))
            scroll.mount(Input(
                value=self._original.get(field_key, ""),
                id=f"field-{field_key}",
                name=field_key,
            ))
        scroll.mount(Button("^W Save", id="save-btn", variant="primary"))

    def save(self) -> None:
        if not self._path:
            return
        tags = {inp.name: inp.value for inp in self.query(Input)}
        write(self._path, tags)
        self._original = dict(tags)
        self.post_message(self.FileSaved(self._path))

    @property
    def is_dirty(self) -> bool:
        if not self._path:
            return False
        return any(
            inp.value != self._original.get(inp.name, "")
            for inp in self.query(Input)
        )

    def on_input_changed(self, _: Input.Changed) -> None:
        if self._path:
            self.post_message(self.DirtyChanged(self._path, self.is_dirty))

    def on_button_pressed(self, event: Button.Pressed) -> None:
        if event.button.id == "save-btn":
            self.save()
```

```python
                    yield Checkbox(
```

```python
                        value=False,
                        id=f"bcb-{field_key}",
                        name=field_key,
                    )
                    yield Input(
                        placeholder=FIELD_LABELS[field_key],
                        id=f"bv-{field_key}",
                        name=field_key,
                        disabled=True,
                    )
            with Horizontal():
                yield Button("Apply", id="batch-apply-btn", variant="primary")
                yield Button("Cancel", id="batch-cancel-btn")

    def on_checkbox_changed(self, event: Checkbox.Changed) -> None:
        if event.checkbox.name:
            inp = self.query_one(f"#bv-{event.checkbox.name}", Input)
            inp.disabled = not event.value

    def on_button_pressed(self, event: Button.Pressed) -> None:
        if event.button.id == "batch-cancel-btn":
            self.dismiss(None)
            return
        result: dict[str, str] = {}
        for field_key in self._active_fields:
            cb = self.query_one(f"#bcb-{field_key}", Checkbox)
            if cb.value:
                inp = self.query_one(f"#bv-{field_key}", Input)
                result[field_key] = inp.value
        self.dismiss(result if result else None)
```

```python
from __future__ import annotations
from pathlib import Path
from textual.app import ComposeResult
from textual.binding import Binding
from textual.screen import Screen, ModalScreen
from textual.widgets import Header, Footer, Button, Label
from textual.containers import Horizontal, Vertical
from widgets.file_list import FileList
from widgets.tag_editor import TagEditor


class _ConfirmScreen(ModalScreen[bool]):
    def compose(self) -> ComposeResult:
        with Vertical():
            yield Label("Unsaved changes — discard and continue?")
            yield Button("Discard", id="discard-btn", variant="error")
            yield Button("Keep editing", id="cancel-btn")

    def on_button_pressed(self, event: Button.Pressed) -> None:
        self.dismiss(event.button.id == "discard-btn")


class MainScreen(Screen):
    BINDINGS = [
        Binding("ctrl+b", "browse", "^B Browse"),
        Binding("ctrl+s", "settings", "^S Settings"),
        Binding("ctrl+w", "save", "^W Save"),
        Binding("ctrl+e", "batch_edit", "^E Batch Edit"),
        Binding("ctrl+q", "quit", "^Q Quit"),
    ]

    def compose(self) -> ComposeResult:
        yield Header()
        with Horizontal():
            yield FileList(id="file-list")
            yield TagEditor(id="tag-editor")
        yield Footer()

    def action_browse(self) -> None:
        from screens.browser_screen import BrowserScreen
        self.app.push_screen(BrowserScreen(), self._on_browse_result)

    def _on_browse_result(self, path: object) -> None:
        if isinstance(path, Path):
            self.query_one(FileList).load(path)

    def action_settings(self) -> None:
        from screens.settings_screen import SettingsScreen
        self.app.push_screen(SettingsScreen(), lambda _: self.query_one(TagEditor).reload_settings())

    def action_save(self) -> None:
        te = self.query_one(TagEditor)
        try:
            te.save()
        except Exception as exc:
            self.app.notify(f"Save failed: {exc}", severity="error")

    def action_batch_edit(self) -> None:
        from screens.batch_edit_screen import BatchEditScreen
        self.app.push_screen(BatchEditScreen(), self._apply_batch)

    def _apply_batch(self, result: dict | None) -> None:
        if not result:
            return
        from core.tag_io import read, write
        failed: list[str] = []
        for path in self.query_one(FileList).all_paths():
            try:
                tags = read(path)
                tags.update(result)
                write(path, tags)
            except Exception:
                failed.append(path.name)
        if failed:
            self.app.notify(f"Failed: {', '.join(failed)}", severity="error")
        else:
            self.app.notify("Batch edit applied")

    def on_file_list_file_selected(self, message: FileList.FileSelected) -> None:
        te = self.query_one(TagEditor)
        if te.is_dirty:
            self.app.push_screen(
                _ConfirmScreen(),
                lambda discard: te.load(message.path) if discard else None,
            )
        else:
            te.load(message.path)

    def on_tag_editor_file_saved(self, message: TagEditor.FileSaved) -> None:
        self.query_one(FileList).mark_clean(message.path)

    def on_tag_editor_dirty_changed(self, message: TagEditor.DirtyChanged) -> None:
        fl = self.query_one(FileList)
        if message.dirty:
            fl.mark_dirty(message.path)
        else:
            fl.mark_clean(message.path)
```

- Editing a field marks the file dirty (`*`) in the Files panel
- `Ctrl+W` (or clicking the Save button) saves and removes the `*`
- Clicking a different file while dirty triggers a discard confirmation dialog
- `Ctrl+E` opens the Batch Edit modal; entering a value and pressing Apply updates all files
- `Ctrl+S` opens Settings; toggling a checkbox and closing re-renders the tag editor with the extra field
- `Ctrl+Q` exits cleanly
